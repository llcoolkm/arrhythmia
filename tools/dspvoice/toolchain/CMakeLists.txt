cmake_minimum_required(VERSION 3.15)
project(dspvoice_assembler LANGUAGES C CXX)
set(CMAKE_CXX_STANDARD 17)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
set(DSP56300_SOURCE "" CACHE PATH "Pinned dsp56300 Git checkout (see UPSTREAM.md)")
find_package(Git REQUIRED)
function(require_revision directory revision)
    execute_process(COMMAND "${GIT_EXECUTABLE}" -C "${directory}" rev-parse HEAD
        RESULT_VARIABLE result OUTPUT_VARIABLE actual OUTPUT_STRIP_TRAILING_WHITESPACE)
    if(NOT result EQUAL 0 OR NOT actual STREQUAL revision)
        message(FATAL_ERROR "${directory}: expected Git revision ${revision}; see UPSTREAM.md")
    endif()
    execute_process(COMMAND "${GIT_EXECUTABLE}" -C "${directory}" status --porcelain --untracked-files=all
        RESULT_VARIABLE result OUTPUT_VARIABLE dirty OUTPUT_STRIP_TRAILING_WHITESPACE)
    if(NOT result EQUAL 0 OR NOT dirty STREQUAL "")
        message(FATAL_ERROR "${directory}: a clean pinned checkout is required")
    endif()
endfunction()
if(NOT IS_DIRECTORY "${DSP56300_SOURCE}/source/dsp56kEmu")
    message(FATAL_ERROR "Set DSP56300_SOURCE to the pinned checkout described in UPSTREAM.md")
endif()
require_revision("${DSP56300_SOURCE}/source/asmjit" "3577608cab0bc509f856ebf6e41b2f9d9f71acc4")
require_revision("${DSP56300_SOURCE}" "2a7ca1e9ce4b0d74b47d9a6384f6d15c1588d2b0")
set(DSP56300_DEBUGGER OFF CACHE BOOL "" FORCE)
set(DSP56K_FORCE_INTERPRETER ON CACHE BOOL "" FORCE)
set(BUILD_TESTING OFF CACHE BOOL "" FORCE)
add_subdirectory("${DSP56300_SOURCE}" upstream EXCLUDE_FROM_ALL)
add_executable(dspasm dspasm.cpp)
target_link_libraries(dspasm PRIVATE dsp56kEmu)
