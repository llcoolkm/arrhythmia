cmake_minimum_required(VERSION 3.15)
project(dsp_harness LANGUAGES C CXX)
set(CMAKE_CXX_STANDARD 17)
set(DSP56300_SOURCE "" CACHE PATH "Your checkout of the pinned dsp56300 commit with both patches applied")
if(NOT EXISTS "${DSP56300_SOURCE}/source/dsp56kEmu/dsp.h")
  message(FATAL_ERROR "Set DSP56300_SOURCE to the patched dsp56300 checkout")
endif()
file(READ "${DSP56300_SOURCE}/source/dsp56kEmu/memory.h" MEMORY_HEADER)
if(NOT MEMORY_HEADER MATCHES "g_memoryAccessHook")
  message(FATAL_ERROR "Apply access-hook.patch to the checkout first")
endif()
file(READ "${DSP56300_SOURCE}/source/dsp56kEmu/peripherals.h" PERIPHERAL_HEADER)
if(PERIPHERAL_HEADER MATCHES "class PeripheralsNop final")
  message(FATAL_ERROR "Apply peripherals-open.patch to the checkout first")
endif()
set(DSP56300_DEBUGGER OFF CACHE BOOL "" FORCE)
set(DSP56K_FORCE_INTERPRETER ON CACHE BOOL "" FORCE)
add_subdirectory("${DSP56300_SOURCE}" upstream EXCLUDE_FROM_ALL)
add_executable(dsp-harness harness.cpp)
target_link_libraries(dsp-harness PRIVATE dsp56kEmu)
