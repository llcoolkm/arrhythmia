cmake_minimum_required(VERSION 3.15)
project(meter_tools LANGUAGES C CXX)
set(CMAKE_CXX_STANDARD 17)
set(DSP56300_SOURCE "" CACHE PATH "upstream")
set(DSP56300_DEBUGGER OFF CACHE BOOL "" FORCE)
set(DSP56K_FORCE_INTERPRETER ON CACHE BOOL "" FORCE)
add_subdirectory("${DSP56300_SOURCE}" upstream EXCLUDE_FROM_ALL)
add_executable(dspasm dspasm.cpp)
target_link_libraries(dspasm PRIVATE dsp56kEmu)
# emulator probe: private tree only (not in the public export)
if(EXISTS "${CMAKE_CURRENT_SOURCE_DIR}/meter_probe.cpp")
    add_executable(meter_probe meter_probe.cpp)
    target_link_libraries(meter_probe PRIVATE dsp56kEmu)
endif()
